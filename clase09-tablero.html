Código completo · repaso2.php

<?php
// Ejercicio 2 — elseif y switch
$costoEstimado = 95000;
$tipoMantenimiento = "correctivo";

// elseif: se evalúa de arriba hacia abajo y solo se ejecuta la PRIMERA rama verdadera
if ($costoEstimado >= 300000) {
    echo "Costo alto<br>";
} elseif ($costoEstimado >= 100000) {
    echo "Costo medio<br>";
} else {
    echo "Costo bajo<br>";
}

// switch: compara $tipoMantenimiento con cada case por igualdad exacta
switch ($tipoMantenimiento) {
    case "preventivo":
        echo "Mantenimiento programado";
        break;                 // break: sale del switch
    case "correctivo":
        echo "Mantenimiento por falla";
        break;
    default:                   // default: si ningún case coincide
        echo "Tipo desconocido";
}
?>

<?php
// Ejercicio 8 — Página única: formulario + validación + INSERT + listado
$conexion = mysqli_connect("localhost", "root", "", "practica_tablero");
if (!$conexion) {
    die("Error de conexión: " . mysqli_connect_error());
}
$mensaje = "";

// Solo se procesa si se presionó el botón (name="enviar")
if (isset($_POST["enviar"])) {
    // trim quita espacios: un campo vacío llega como "" aunque isset dé true
    if (!isset($_POST["equipo"]) || trim($_POST["equipo"]) === "") {
        $mensaje = "Falta el equipo. No se guardó nada.";
    } else {
        // Textos: se escapan; números: se convierten
        $equipo = mysqli_real_escape_string($conexion, trim($_POST["equipo"]));
        $tipo = mysqli_real_escape_string($conexion, $_POST["tipoMantenimiento"]);
        $horas = (float) $_POST["horasInvertidas"];
        $costo = (float) $_POST["costoEstimado"];
        // Checkbox: si no se marca, no llega; isset decide 1 o 0
        $completado = isset($_POST["completado"]) ? 1 : 0;
        $sql = "INSERT INTO repaso_mantenimientos (equipo, tipoMantenimiento, completado, horasInvertidas, costoEstimado)
                VALUES ('$equipo', '$tipo', $completado, $horas, $costo)";
        mysqli_query($conexion, $sql);             // con INSERT devuelve true
        $mensaje = "Guardado con id: " . mysqli_insert_id($conexion);
    }
}
